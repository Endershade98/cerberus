# tests/unit/domain/shared/test_aggregate_root.py

from dataclasses import dataclass

from domain.shared.aggregate_root import AggregateRoot
from domain.shared.domain_event import DomainEvent


@dataclass(frozen=True)
class FakeEvent(DomainEvent):
    value: str = "test"


def test_aggregate_root_collects_domain_events():
    aggregate = AggregateRoot()

    event = FakeEvent()

    aggregate.add_event(event)

    events = aggregate.pull_events()

    assert events == [event]


def test_pull_events_clears_event_collection():
    aggregate = AggregateRoot()

    aggregate.add_event(FakeEvent())

    first_pull = aggregate.pull_events()
    second_pull = aggregate.pull_events()

    assert len(first_pull) == 1
    assert second_pull == []


def test_aggregate_root_can_collect_multiple_events():
    aggregate = AggregateRoot()

    first = FakeEvent(value="first")
    second = FakeEvent(value="second")

    aggregate.add_event(first)
    aggregate.add_event(second)

    assert aggregate.pull_events() == [first, second]

    def save(self, member):

    def get(self, member_id):
        return self.store.get(member_id.value)

    def find_by_email(self, email):
        return next(
            (m for m in self.store.values() if m.email == email),

        )

    def __init__(self, store):

    def save(self, record):

    def get_all(self):

        self.energy_repository = FakeEnergyRepo(self.energy)

    def commit(self):
        pass

    def rollback(self):
        self.members.clear()
        self.energy.clear()
        self.events.clear()
